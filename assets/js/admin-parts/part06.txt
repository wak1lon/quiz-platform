async function requireLoginIfNeeded(){
  if(!firebaseEnabled()) return true;
  const user=await waitForAuth();
  if(user) return true;
  modalRoot.innerHTML=`<div class="modal-backdrop"><section class="modal" style="max-width:440px"><div class="modal-head"><div><strong>Entrar no QuizPlatform</strong><div class="muted" style="font-size:12px">Firebase Authentication está ativo</div></div></div><form id="loginForm" class="modal-body"><div class="field"><label>Email</label><input id="loginEmail" type="email" required autocomplete="username"></div><div class="field"><label>Senha</label><input id="loginPassword" type="password" required autocomplete="current-password"></div><p id="loginError" class="muted" style="font-size:12px"></p><button class="btn btn-primary" style="width:100%" type="submit">Entrar</button></form></section></div>`;
  return await new Promise(resolve=>{document.getElementById('loginForm').onsubmit=async e=>{e.preventDefault();const err=document.getElementById('loginError');err.textContent='';try{await loginEmail(document.getElementById('loginEmail').value.trim(),document.getElementById('loginPassword').value);modalRoot.innerHTML='';resolve(true);}catch(ex){err.textContent='Não foi possível entrar. Confira email, senha e se Email/Senha está habilitado no Firebase.';console.error(ex);}};});
}
