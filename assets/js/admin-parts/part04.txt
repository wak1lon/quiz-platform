function toggleQ(label,path,checked){return `<div class="field-row"><span>${label}</span><label class="toggle"><input type="checkbox" data-qpath="${path}" ${checked?'checked':''}><span></span></label></div>`;}
function bindQuestionEditorEvents(){
  const canvas=document.getElementById('canvasDrop');let dragFieldId=null;
  document.querySelectorAll('[data-palette-type]').forEach(el=>{el.addEventListener('click',()=>addField(el.dataset.paletteType));el.addEventListener('dragstart',e=>{e.dataTransfer.setData('application/x-palette',el.dataset.paletteType);});});
  canvas.addEventListener('dragover',e=>{e.preventDefault();canvas.classList.add('dragover');});canvas.addEventListener('dragleave',()=>canvas.classList.remove('dragover'));canvas.addEventListener('drop',e=>{e.preventDefault();canvas.classList.remove('dragover');const type=e.dataTransfer.getData('application/x-palette');if(type){addField(type);return;}const moving=e.dataTransfer.getData('application/x-field');const target=e.target.closest('[data-field-id]')?.dataset.fieldId;if(moving&&target&&moving!==target){const from=editorQuiz.questions.findIndex(q=>q.id===moving),to=editorQuiz.questions.findIndex(q=>q.id===target);const [item]=editorQuiz.questions.splice(from,1);editorQuiz.questions.splice(to,0,item);renderQuestionEditor();}});
  canvas.querySelectorAll('[data-field-id]').forEach(el=>{el.addEventListener('click',()=>{selectedFieldId=el.dataset.fieldId;renderQuestionEditor();});el.addEventListener('dragstart',e=>{dragFieldId=el.dataset.fieldId;e.dataTransfer.setData('application/x-field',dragFieldId);});});
  const q=editorQuiz.questions.find(x=>x.id===selectedFieldId);if(!q)return;document.getElementById('propsPanel').querySelectorAll('[data-qpath]').forEach(el=>{const event=el.type==='checkbox'||el.tagName==='SELECT'?'change':'input';el.addEventListener(event,()=>setPath(q,el.dataset.qpath,valueFromInput(el)));});
  const optionsEditor=document.getElementById('optionsEditor');if(optionsEditor)optionsEditor.addEventListener('input',()=>{q.options=optionsEditor.value.split('\n').map(line=>line.trim()).filter(Boolean).map((line,i)=>{const [label,value,weight,icon,image]=line.split('|');return{id:q.options?.[i]?.id||uid('opt'),label:label?.trim()||`Opção ${i+1}`,value:(value?.trim()||slugify(label)||`opcao_${i+1}`),weight:Number(weight)||0,icon:icon?.trim()||'',image:image?.trim()||''};});});
