function customQuizUrl(q){
  const custom=safeSettingsApi.getQuizDomain(q.id);
  if(!custom)return '';
  try{
    const u=new URL(custom);
    if(!u.searchParams.has('slug'))u.searchParams.set('slug',q.slug);
    return u.href;
  }catch{return '';}
}
