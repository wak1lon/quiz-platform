async function applyAuthenticatedAvatar(){
  const avatar=document.getElementById('accountAvatar')||document.querySelector('.avatar');
  if(!avatar)return;
  const localProfile=safeSettingsApi.getActiveProfile();
  let user=null;
  try{user=await accountAuth.currentUser();}catch{}
  const photo=localProfile?.photoData||user?.photoURL||'';
  const name=localProfile?.name||user?.displayName||'Wakilon Gestor';
  if(photo)avatar.innerHTML=`<img src="${escapeHtml(photo)}" alt="Foto de ${escapeHtml(name)}" style="width:100%;height:100%;object-fit:cover;border-radius:50%">`;
  else avatar.textContent=name.split(/\s+/).map(x=>x[0]).join('').slice(0,2).toUpperCase();
  avatar.title='Abrir menu da conta';
}
