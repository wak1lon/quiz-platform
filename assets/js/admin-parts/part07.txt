const visualMemory=await import(new URL('platform-memory.js',__base).href);
function applyPlatformBrand(){const m=visualMemory.applyPlatformMemory();const brand=document.querySelector('.brand strong');if(brand)brand.textContent=m.platformName||'QuizPlatform';const mark=document.querySelector('.brand-mark');if(mark){if(m.logoData)mark.innerHTML=`<img src="${m.logoData}" alt="Logo" style="width:100%;height:100%;object-fit:contain;border-radius:inherit">`;else mark.textContent=(m.platformName||'Q').trim().charAt(0).toUpperCase()||'Q';}return m;}
applyPlatformBrand();
