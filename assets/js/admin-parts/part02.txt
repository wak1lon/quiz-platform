function renderEditor(){
  if(!editorQuiz)editorQuiz=blankQuiz();
  content.innerHTML=`<div class="editor-topline"><input id="editorTitle" value="${escapeHtml(editorQuiz.title)}" aria-label="Título do quiz"><div class="actions"><button class="btn btn-secondary" id="saveDraft">Salvar</button><button class="btn btn-primary" id="publishQuiz">Publicar</button></div></div><div class="tabs">${[['config','Configurações'],['questions','Perguntas'],['design','Design'],['integrations','Integrações'],['results','Resultados']].map(([id,label])=>`<button class="tab ${editorTab===id?'active':''}" data-editor-tab="${id}">${label}</button>`).join('')}</div><div id="editorBody"></div>`;
  document.getElementById('editorTitle').addEventListener('input',e=>{editorQuiz.title=e.target.value;if(!editorQuiz.slug||editorQuiz.slug.startsWith('quiz-'))editorQuiz.slug=uniqueSlug(e.target.value,editorQuiz.id);});
  document.querySelectorAll('[data-editor-tab]').forEach(b=>b.onclick=()=>{editorTab=b.dataset.editorTab;renderEditor();});
  document.getElementById('saveDraft').onclick=()=>saveCurrent(false);document.getElementById('publishQuiz').onclick=()=>saveCurrent(true);
  if(editorTab==='questions')renderQuestionEditor();else if(editorTab==='config')renderConfigTab();else if(editorTab==='design')renderDesignTab();else if(editorTab==='integrations')renderIntegrationsTab();else renderResultTab();
}

function bindObjectForm(root,rerender=false){root.querySelectorAll('[data-path]').forEach(el=>{const handler=()=>{setPath(editorQuiz,el.dataset.path,valueFromInput(el));if(el.dataset.path==='title'&&!editorQuiz.slug)editorQuiz.slug=uniqueSlug(editorQuiz.title,editorQuiz.id);if(rerender)renderEditor();};el.addEventListener(el.type==='checkbox'||el.tagName==='SELECT'?'change':'input',handler);});}
