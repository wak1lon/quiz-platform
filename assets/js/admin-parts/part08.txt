async function renderSafeSettings(){
  const m=visualMemory.getPlatformMemory();const history=visualMemory.getPlatformHistory();
  content.innerHTML=`<div class="section-head"><div><h2>Identidade visual</h2><p>Memória local isolada. Esta tela não modifica Firebase nem integrações externas.</p></div><span class="status-pill success">Memória segura</span></div><div class="settings-grid"><section class="card settings-card"><h3>Marca e aparência</h3>${visualMemoryFields(m)}<div class="actions" style="margin-top:18px"><button class="btn btn-primary" id="saveMemory">Salvar identidade</button><button class="btn btn-secondary" id="undoMemory" ${history.length?'':'disabled'}>Desfazer última</button><button class="btn btn-ghost" id="resetMemory">Restaurar padrão</button></div></section><section class="card settings-card"><h3>Memória e segurança</h3><div class="code-box">Armazenamento: navegador local · sem Firestore</div><p class="muted" style="font-size:12px">Esta memória aceita somente nome, logo, favicon, cores, fonte, bordas e aparência. Firebase, autenticação, GTM, Pixel, Analytics, webhook e WhatsApp ficam fora dela.</p><div class="actions" style="margin-top:14px"><button class="btn btn-secondary" id="exportMemory">Exportar backup</button><label class="btn btn-secondary" style="cursor:pointer">Importar backup<input id="importMemory" type="file" accept="application/json" hidden></label></div><h3 style="margin-top:22px">Histórico</h3><div class="list">${history.length?history.slice(0,5).map((h,i)=>`<div class="list-row"><div><strong>${escapeHtml(h.platformName||'QuizPlatform')}</strong><small>${h.updatedAt?formatDate(h.updatedAt):'Versão anterior'}</small></div><button class="btn btn-sm btn-ghost" data-restore-memory="${i}">Restaurar</button></div>`).join(''):'<div class="empty">O histórico aparecerá após a primeira alteração.</div>'}</div><div class="card" style="padding:16px;margin-top:18px;background:var(--background)"><strong>Área crítica protegida</strong><p class="muted" style="font-size:12px;margin-bottom:0">A identidade visual não grava nem altera os vínculos externos da plataforma.</p></div></section></div>`;
  document.getElementById('saveMemory').onclick=async()=>{try{const patch=collectVisualMemory();const logo=document.getElementById('memLogo').files[0],favicon=document.getElementById('memFavicon').files[0];if(logo)patch.logoData=await visualMemory.fileToDataUrl(logo);if(favicon)patch.faviconData=await visualMemory.fileToDataUrl(favicon);visualMemory.savePlatformMemory(patch);applyPlatformBrand();toast('Identidade salva somente na memória local.','success');renderSafeSettings();}catch(e){toast(e.message||'Não foi possível salvar.','error');}};
  document.getElementById('undoMemory').onclick=()=>{if(visualMemory.restorePlatformMemory(0)){applyPlatformBrand();toast('Última versão restaurada.','success');renderSafeSettings();}};
  document.getElementById('resetMemory').onclick=()=>{if(confirm('Restaurar a identidade visual padrão?')){visualMemory.resetPlatformMemory();applyPlatformBrand();toast('Identidade padrão restaurada.','success');renderSafeSettings();}};
  document.getElementById('exportMemory').onclick=()=>download('quizplatform-identidade-visual.json',visualMemory.exportPlatformMemory(),'application/json');
  document.getElementById('importMemory').onchange=async e=>{try{const file=e.target.files[0];if(!file)return;visualMemory.importPlatformMemory(await file.text());applyPlatformBrand();toast('Backup visual importado.','success');renderSafeSettings();}catch(err){toast(err.message||'Backup inválido.','error');}};
  content.querySelectorAll('[data-restore-memory]').forEach(btn=>btn.onclick=()=>{visualMemory.restorePlatformMemory(Number(btn.dataset.restoreMemory));applyPlatformBrand();toast('Versão restaurada.','success');renderSafeSettings();});
  ['memName','memPrimary','memSecondary','memBackground','memSurface','memText','memAccent','memFont','memRadius'].forEach(id=>document.getElementById(id).addEventListener('input',()=>{visualMemory.applyPlatformMemory({...m,...collectVisualMemory()});applyPlatformBrand();}));
}
renderSettings=renderSafeSettings;
